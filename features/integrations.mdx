### GitHub Actions

Redact documents in your CI/CD pipeline:

```yaml
name: Redact Documents
on:
  push:
    paths:
      - 'documents/**'

jobs:
  redact:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      
      - name: Redact Documents
        uses: nvisy/redact-action@v1
        with:
          api-key: ${{ secrets.NVISY_API_KEY }}
          files: 'documents/**/*.pdf'
          detection-types: 'pii,phi'
          
      - name: Upload Redacted
        uses: actions/upload-artifact@v3
        with:
          name: redacted-documents
          path: redacted/
```

### AWS S3 Integration

Automatically redact documents uploaded to S3:

```typescript
// AWS Lambda function
import { S3Event } from 'aws-lambda';
import { S3 } from '@aws-sdk/client-s3';
import { NvisyClient } from '@nvisy/sdk';

const s3 = new S3();
const nvisy = new NvisyClient({
  apiKey: process.env.NVISY_API_KEY
});

export const handler = async (event: S3Event) => {
  for (const record of event.Records) {
    const bucket = record.s3.bucket.name;
    const key = record.s3.object.key;
    
    // Download from S3
    const object = await s3.getObject({ Bucket: bucket, Key: key });
    
    // Redact document
    const result = await nvisy.documents.redact({
      file: object.Body,
      detectionTypes: ['pii', 'phi', 'financial']
    });
    
    // Upload redacted version
    const redactedKey = `redacted/${key}`;
    await s3.putObject({
      Bucket: bucket,
      Key: redactedKey,
      Body: await result.getRedactedBuffer()
    });
  }
};
```

### Google Drive Integration

Use Google Drive API with Nvisy:

```typescript
import { google } from 'googleapis';
import { NvisyClient } from '@nvisy/sdk';

const drive = google.drive('v3');
const nvisy = new NvisyClient({
  apiKey: process.env.NVISY_API_KEY
});

async function redactDriveDocument(fileId: string) {
  // Download from Drive
  const response = await drive.files.get(
    { fileId, alt: 'media' },
    { responseType: 'stream' }
  );
  
  // Redact document
  const result = await nvisy.documents.redact({
    file: response.data,
    detectionTypes: ['pii', 'phi']
  });
  
  // Upload redacted version
  await drive.files.create({
    requestBody: {
      name: 'Redacted Document',
      parents: [folderId]
    },
    media: {
      body: await result.getRedactedBuffer()
    }
  });
}
```

### Slack Integration

Create a Slack bot that redacts documents:

```typescript
import { App } from '@slack/bolt';
import { NvisyClient } from '@nvisy/sdk';

const app = new App({
  token: process.env.SLACK_BOT_TOKEN,
  signingSecret: process.env.SLACK_SIGNING_SECRET
});

const nvisy = new NvisyClient({
  apiKey: process.env.NVISY_API_KEY
});

// Listen for file uploads
app.event('file_shared', async ({ event, client }) => {
  const fileInfo = await client.files.info({ file: event.file_id });
  const file = fileInfo.file;
  
  // Download file
  const response = await fetch(file.url_private_download, {
    headers: { Authorization: `Bearer ${process.env.SLACK_BOT_TOKEN}` }
  });
  
  // Redact document
  const result = await nvisy.documents.redact({
    file: Buffer.from(await response.arrayBuffer()),
    detectionTypes: ['pii', 'phi']
  });
  
  // Upload redacted version
  await client.files.upload({
    channels: event.channel_id,
    file: await result.getRedactedBuffer(),
    filename: `redacted_${file.name}`,
    initial_comment: 'Redacted version uploaded'
  });
});

await app.start(3000);
```

### SharePoint Integration

```typescript
import { Client } from '@microsoft/microsoft-graph-client';
import { NvisyClient } from '@nvisy/sdk';

const graphClient = Client.init({
  authProvider: // Your auth provider
});

const nvisy = new NvisyClient({
  apiKey: process.env.NVISY_API_KEY
});

async function redactSharePointDocument(driveId: string, itemId: string) {
  // Download from SharePoint
  const stream = await graphClient
    .api(`/drives/${driveId}/items/${itemId}/content`)
    .getStream();
  
  // Redact document
  const result = await nvisy.documents.redact({
    file: stream,
    detectionTypes: ['pii', 'phi']
  });
  
  // Upload redacted version
  await graphClient
    .api(`/drives/${driveId}/items/${itemId}/content`)
    .put(await result.getRedactedBuffer());
}
```

### Salesforce Integration

```typescript
import jsforce from 'jsforce';
import { NvisyClient } from '@nvisy/sdk';

const conn = new jsforce.Connection({
  loginUrl: 'https://login.salesforce.com'
});

const nvisy = new NvisyClient({
  apiKey: process.env.NVISY_API_KEY
});

async function redactSalesforceAttachment(attachmentId: string) {
  // Download attachment
  const attachment = await conn.sobject('Attachment')
    .retrieve(attachmentId);
  const body = await conn.request(attachment.Body);
  
  // Redact document
  const result = await nvisy.documents.redact({
    file: Buffer.from(body, 'base64'),
    detectionTypes: ['pii', 'financial']
  });
  
  // Create redacted attachment
  await conn.sobject('Attachment').create({
    Name: `Redacted_${attachment.Name}`,
    Body: (await result.getRedactedBuffer()).toString('base64'),
    ParentId: attachment.ParentId
  });
}
```

### Datadog Integration

Send metrics to Datadog:

```typescript
import { StatsD } from 'hot-shots';
import { NvisyClient } from '@nvisy/sdk';

const statsd = new StatsD({
  host: 'localhost',
  port: 8125,
  prefix: 'nvisy.'
});

const nvisy = new NvisyClient({
  apiKey: process.env.NVISY_API_KEY
});

async function redactWithMetrics(file: string) {
  const startTime = Date.now();
  
  try {
    const result = await nvisy.documents.redact({
      file,
      detectionTypes: ['pii']
    });
    
    statsd.increment('redactions.success');
    statsd.histogram('redactions.duration', Date.now() - startTime);
    statsd.gauge('detections.count', result.detections.length);
    
    return result;
  } catch (error) {
    statsd.increment('redactions.error');
    throw error;
  }
}
```

## Integration Examples

### Complete Workflow: S3 → Redact → Notify

```typescript
import { S3Event } from 'aws-lambda';
import { S3 } from '@aws-sdk/client-s3';
import { NvisyClient } from '@nvisy/sdk';
import { SNS } from '@aws-sdk/client-sns';

const s3 = new S3();
const sns = new SNS();
const nvisy = new NvisyClient({
  apiKey: process.env.NVISY_API_KEY
});

export const handler = async (event: S3Event) => {
  for (const record of event.Records) {
    const bucket = record.s3.bucket.name;
    const key = record.s3.object.key;
    
    try {
      // 1. Download from S3
      const object = await s3.getObject({ Bucket: bucket, Key: key });
      
      // 2. Redact document
      const result = await nvisy.documents.redact({
        file: object.Body,
        detectionTypes: ['pii', 'phi', 'financial']
      });
      
      // 3. Upload redacted version
      const redactedKey = `redacted/${key}`;
      await s3.putObject({
        Bucket: bucket,
        Key: redactedKey,
        Body: await result.getRedactedBuffer()
      });
      
      // 4. Send notification
      await sns.publish({
        TopicArn: process.env.SNS_TOPIC_ARN,
        Subject: 'Document Redacted',
        Message: JSON.stringify({
          originalFile: key,
          redactedFile: redactedKey,
          detectionsCount: result.detections.length
        })
      });
      
    } catch (error) {
      console.error('Redaction failed:', error);
      
      // Send error notification
      await sns.publish({
        TopicArn: process.env.SNS_TOPIC_ARN,
        Subject: 'Redaction Failed',
        Message: `Failed to redact ${key}: ${error.message}`
      });
    }
  }
};
```
